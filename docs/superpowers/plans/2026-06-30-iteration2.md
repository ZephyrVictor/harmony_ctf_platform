> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implement 6 features for the OpenHarmony CTF platform iteration 2: 0day submission info tab, appeal/0day frontend closure, team score management, ban/alert system, flag batch operations, and level name display.

**Architecture:** Backend changes follow existing GZCTF patterns (`[ApiController]` + `[Route("api/...")]` + constructor injection + `[RequireRole]`/`[RequireUser]` attributes, EF Core models with `[MemoryPackable]`). Frontend follows existing React + Mantine + react-i18next patterns. Each feature is a vertical slice: backend model → migration → controller → API test (curl) → frontend page → i18n → manual verify.

**Tech Stack:** .NET 8 + EF Core + PostgreSQL (backend), React 18 + Mantine 7 + react-i18next + Vite (frontend), curl (backend API testing), Playwright via `drive.js` (frontend e2e, covered in test plan).

```tsx
  const [markdown, setMarkdown] = useState<string>('')

        </Text>

        ) : (

  )
}
```

### Task 3.2: ScoreAdjustmentController

**Files:**
- Create: `src/GZCTF/Controllers/ScoreAdjustmentController.cs`

- [ ] **Step 1: 创建 Controller**

```csharp
using System.Net.Mime;
using System.Security.Claims;
using GZCTF.Middlewares;
using GZCTF.Models.Data;
using GZCTF.Models.Request.Edit;
using GZCTF.Models.Response;
using GZCTF.Repositories.Interface;
using Microsoft.AspNetCore.Mvc;
using Microsoft.EntityFrameworkCore;

namespace GZCTF.Controllers;

[ApiController]
[Route("api/admin/games/{gameId}/score-adjustments")]
[Produces(MediaTypeNames.Application.Json)]
[ProducesResponseType(typeof(RequestResponse), StatusCodes.Status400BadRequest)]
[ProducesResponseType(typeof(RequestResponse), StatusCodes.Status401Unauthorized)]
[ProducesResponseType(typeof(RequestResponse), StatusCodes.Status403Forbidden)]
public class ScoreAdjustmentController(
    AppDbContext context,
    IGameRepository gameRepository,
    ITeamRepository teamRepository) : ControllerBase
{
    /// <summary>创建调分记录（手动或 0day 加分）</summary>
    [RequireRole(Role.Admin)]
    [HttpPost]
    [ProducesResponseType(typeof(ScoreAdjustmentDto), StatusCodes.Status201Created)]
    public async Task<IActionResult> Create(
        [FromRoute] int gameId,
        [FromBody] ScoreAdjustmentCreateModel model,
        CancellationToken token)
    {
        if (model.Delta == 0)
            return BadRequest(new RequestResponse("Delta cannot be zero"));

        if (string.IsNullOrWhiteSpace(model.Reason))
            return BadRequest(new RequestResponse("Reason is required"));

        var game = await gameRepository.GetGameById(gameId, token);
        if (game is null)
            return NotFound(new RequestResponse("Game not found"));

        var team = await teamRepository.GetTeamById(model.TeamId, token);
        if (team is null)
            return NotFound(new RequestResponse("Team not found"));

        var userId = User.FindFirstValue(ClaimTypes.NameIdentifier);
        if (userId is null)
            return Unauthorized(new RequestResponse("User not found"));

        var adjustment = new ScoreAdjustment
        {
            Id = Guid.NewGuid(),
            GameId = gameId,
            TeamId = model.TeamId,
            Delta = model.Delta,
            Reason = model.Reason,
            Category = model.Category,
            RelatedInfo = model.RelatedInfo,
            OperatorId = Guid.Parse(userId),
            CreatedAt = DateTime.UtcNow
        };

        context.ScoreAdjustments.Add(adjustment);

        // 写入 GameNotice 通报（全员可见）
        var notice = new GameNotice
        {
            GameId = gameId,
            Type = NoticeType.ScoreAdjustment,
            Values = new string[]
            {
                team.Name,
                model.Delta > 0 ? $"+{model.Delta}" : model.Delta.ToString(),
                model.Reason,
                User.Identity?.Name ?? "admin"
            },
            PublishTimeUtc = DateTime.UtcNow
        };
        context.GameNotices.Add(notice);

        await context.SaveChangesAsync(token);

        return CreatedAtAction(nameof(GetById), new { gameId, id = adjustment.Id }, adjustment);
    }

    /// <summary>获取单条调分记录</summary>
    [RequireRole(Role.Admin)]
    [HttpGet("{id}")]
    [ProducesResponseType(typeof(ScoreAdjustmentDto), StatusCodes.Status200OK)]
    public async Task<IActionResult> GetById([FromRoute] int gameId, [FromRoute] Guid id, CancellationToken token)
    {
        var adj = await context.ScoreAdjustments
            .AsNoTracking()
            .FirstOrDefaultAsync(a => a.GameId == gameId && a.Id == id, token);
        if (adj is null) return NotFound();
        return Ok(adj);
    }

    /// <summary>调分记录列表（分页 + 过滤）</summary>
    [RequireRole(Role.Admin)]
    [HttpGet]
    [ProducesResponseType(typeof(PagedResponse<ScoreAdjustmentDto>), StatusCodes.Status200OK)]
    public async Task<IActionResult> List(
        [FromRoute] int gameId,
        [FromQuery] ScoreAdjustmentCategory? category,
        [FromQuery] Guid? teamId,
        [FromQuery] int page = 1,
        [FromQuery] int pageSize = 20,
        CancellationToken token = default)
    {
        var query = context.ScoreAdjustments
            .AsNoTracking()
            .Where(a => a.GameId == gameId);

        if (category.HasValue) query = query.Where(a => a.Category == category);
        if (teamId.HasValue) query = query.Where(a => a.TeamId == teamId);

        var total = await query.CountAsync(token);
        var items = await query
            .OrderByDescending(a => a.CreatedAt)
            .Skip((page - 1) * pageSize)
            .Take(pageSize)
            .Select(a => new ScoreAdjustmentDto
            {
                Id = a.Id,
                GameId = a.GameId,
                TeamId = a.TeamId,
                Delta = a.Delta,
                Reason = a.Reason,
                Category = a.Category,
                RelatedInfo = a.RelatedInfo,
                OperatorId = a.OperatorId,
                CreatedAt = a.CreatedAt
            })
            .ToListAsync(token);

        return Ok(new PagedResponse<ScoreAdjustmentDto>(items, total));
    }

    /// <summary>搜索队伍（队名/成员/Token 模糊搜）</summary>
    [RequireRole(Role.Admin)]
    [HttpGet("/api/admin/games/{gameId}/teams/search")]
    [ProducesResponseType(typeof(TeamSearchResultDto[]), StatusCodes.Status200OK)]
    public async Task<IActionResult> SearchTeams(
        [FromRoute] int gameId,
        [FromQuery] string q,
        CancellationToken token)
    {
        if (string.IsNullOrWhiteSpace(q))
            return Ok(Array.Empty<TeamSearchResultDto>());

        var keyword = q.Trim();
        var teams = await context.Teams
            .AsNoTracking()
            .Where(t => t.Games!.Any(g => g.Id == gameId))
            .Where(t =>
                EF.Functions.ILike(t.Name, $"%{keyword}%") ||
                EF.Functions.ILike(t.TeamToken, $"%{keyword}%") ||
                t.Members!.Any(m =>
                    EF.Functions.ILike(m.User!.UserName!, $"%{keyword}%")))
            .Select(t => new TeamSearchResultDto
            {
                TeamId = t.Id,
                TeamName = t.Name,
                TeamToken = t.TeamToken,
                Members = t.Members!.Select(m => m.User!.UserName!).ToArray()
            })
            .Take(20)
            .ToListAsync(token);

        return Ok(teams);
    }
}
```

```csharp
using System.Net.Mime;
using System.Security.Claims;

[Produces(MediaTypeNames.Application.Json)]
[ProducesResponseType(typeof(RequestResponse), StatusCodes.Status400BadRequest)]
[ProducesResponseType(typeof(RequestResponse), StatusCodes.Status401Unauthorized)]
[ProducesResponseType(typeof(RequestResponse), StatusCodes.Status403Forbidden)]
public class BanController(AppDbContext context) : ControllerBase
{
    [RequireRole(Role.Admin)]
    [HttpGet]
    [ProducesResponseType(typeof(BanStateDto[]), StatusCodes.Status200OK)]
    public async Task<IActionResult> List(
        [FromRoute] int gameId,
        [FromQuery] Guid? teamId,
        [FromQuery] Guid? trackId,
        CancellationToken token)
    {
        var query = context.ChallengeTrackLevelBanStates
            .AsNoTracking()
            .Where(b => b.Track!.Challenge!.GameId == gameId);

        if (teamId.HasValue) query = query.Where(b => b.TeamId == teamId);
        if (trackId.HasValue) query = query.Where(b => b.TrackId == trackId);

        var bans = await query
            .OrderByDescending(b => b.BannedAt)
            .Select(b => new BanStateDto
            {
                Id = b.Id,
                TeamId = b.TeamId,
                TrackId = b.TrackId,
                Level = b.Level,
                Reason = b.Reason,
                BannedBy = b.BannedBy,
                IsManual = b.IsManual,
                BannedAt = b.BannedAt
            })
            .ToListAsync(token);

        return Ok(bans);
    }

    [RequireRole(Role.Admin)]
    [HttpPost]
    [ProducesResponseType(typeof(BanStateDto), StatusCodes.Status201Created)]
    public async Task<IActionResult> Create(
        [FromRoute] int gameId,
        [FromBody] BanCreateModel model,
        CancellationToken token)
    {
        if (string.IsNullOrWhiteSpace(model.Reason))
            return BadRequest(new RequestResponse("Reason is required"));

        var userId = User.FindFirstValue(ClaimTypes.NameIdentifier);
        if (userId is null) return Unauthorized();

        var ban = new ChallengeTrackLevelBanState
        {
            Id = Guid.NewGuid(),
            TeamId = model.TeamId,
            TrackId = model.TrackId,
            Level = model.Level,
            BanAnswer = string.Empty, // 手动封不设 trap answer
            BannedAt = DateTime.UtcNow,
            Reason = model.Reason,
            BannedBy = Guid.Parse(userId),
            IsManual = true
        };

        context.ChallengeTrackLevelBanStates.Add(ban);
        await context.SaveChangesAsync(token);

        return Ok(ban);
    }

    [RequireRole(Role.Admin)]
    [HttpDelete("{banId}")]
    [ProducesResponseType(StatusCodes.Status204NoContent)]
    public async Task<IActionResult> Delete([FromRoute] int gameId, [FromRoute] Guid banId, CancellationToken token)
    {
        var ban = await context.ChallengeTrackLevelBanStates
            .FirstOrDefaultAsync(b => b.Id == banId, token);
        if (ban is null) return NotFound();

        context.ChallengeTrackLevelBanStates.Remove(ban);
        await context.SaveChangesAsync(token);
        return NoContent();
    }
}
```

```csharp
OnRejected = async (context, cancellationToken) =>
{
    var httpContext = context.HttpContext;
    var teamId = httpContext.User.FindFirstValue("TeamId");
    var gameIdStr = httpContext.GetRouteValue("id")?.ToString();
    var challengeIdStr = httpContext.GetRouteValue("chalId")?.ToString();

    if (Guid.TryParse(teamId, out var tid) && int.TryParse(gameIdStr, out var gameId))
    {
        var db = httpContext.RequestServices.GetRequiredService<AppDbContext>();
        var team = await db.Teams.FindAsync(tid);
        var challenge = challengeIdStr != null
            ? await db.Challenges.FindAsync(int.Parse(challengeIdStr))
            : null;

        // 去重：5 分钟内同一队同一题只记一条
        var since = DateTime.UtcNow.AddMinutes(-5);
        var exists = await db.GameNotices.AnyAsync(n =>
            n.GameId == gameId &&
            n.Type == NoticeType.RateLimitAlert &&
            n.PublishTimeUtc > since, cancellationToken);

        if (!exists)
        {
            db.GameNotices.Add(new GameNotice
            {
                GameId = gameId,
                Type = NoticeType.RateLimitAlert,
                Values = new[]
                {
                    team?.Name ?? "unknown",
                    challenge?.Title ?? "unknown",
                    DateTime.UtcNow.ToString("o")
                },
                PublishTimeUtc = DateTime.UtcNow
            });
            await db.SaveChangesAsync(cancellationToken);
        }
    }

    httpContext.Response.StatusCode = StatusCodes.Status429TooManyRequests;
    await httpContext.Response.WriteAsJsonAsync(
        new RequestResponse("请求过于频繁，请稍后再试"), cancellationToken);
};
```

```csharp
        f.IsEnabled = model.IsEnabled;

    await context.SaveChangesAsync(token);
    return Ok(new RequestResponse($"Updated {flags.Count} flags"));
}

[RequireRole(Role.Admin)]
[HttpPost("flags/batch-delete")]
[ProducesResponseType(typeof(RequestResponse), StatusCodes.Status200OK)]
public async Task<IActionResult> BatchDeleteFlags(
    [FromBody] BatchFlagDeleteModel model,
    CancellationToken token)
{
    if (model.FlagIds.Count == 0)
        return BadRequest(new RequestResponse("No flags selected"));

    var flags = await context.ChallengeFlags
        .Where(f => model.FlagIds.Contains(f.Id))
        .ToListAsync(token);

    context.ChallengeFlags.RemoveRange(flags);
    await context.SaveChangesAsync(token);
    return Ok(new RequestResponse($"Deleted {flags.Count} flags"));
}
```

请求模型：

```csharp

{
    public List<Guid> FlagIds { get; set; } = new();
    public bool IsEnabled { get; set; }
}

public class BatchFlagDeleteModel
{
    public List<Guid> FlagIds { get; set; } = new();
}
```
